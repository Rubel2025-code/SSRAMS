# Gemini API client (SRS §2.1.1). This is the CURRENT Google GenAI SDK,
# imported as `from google import genai`. Only
# apps/ai_advisor/services/gemini_service.py imports it — no other module
# in the project touches the Gemini SDK.
#
# Do NOT add the pre-1.0 Google generative client alongside this one. It
# is end-of-life (its import prints "All support ... has ended"), it uses
# an incompatible call style, and having both installed is what previously
# let gemini_service.py be written against one SDK while the other was the
# one actually present — the AI Advisor then failed closed on every
# request. apps/ai_advisor/tests.py has a guard test for this.
#
# Upper bound is deliberate: this SDK ships breaking changes on major
# version bumps, and the whole integration is one module, so pinning the
# major is cheap insurance.
google-genai>=1.0,<3

# WSGI server for running the app outside `manage.py runserver` in a
# staging/production-like environment.
gunicorn>=22.0
